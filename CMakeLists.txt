cmake_minimum_required(VERSION 3.16)
project(OpenBabelCN LANGUAGES C CXX)
if(WIN32)
  enable_language(RC)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENBABEL_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/openbabel" CACHE PATH "Open Babel 3.2.1 source")
set(BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(WITH_STATIC_INCHI ON CACHE BOOL "" FORCE)
set(WITH_INCHI ON CACHE BOOL "" FORCE)
set(BUILD_GUI OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_MAEPARSER OFF CACHE BOOL "" FORCE)
set(WITH_COORDGEN OFF CACHE BOOL "" FORCE)
set(WITH_JSON OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_LibXml2 TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_ZLIB TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Cairo TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_Boost TRUE)
add_subdirectory("${OPENBABEL_SOURCE_DIR}" openbabel)
target_include_directories(openbabel PRIVATE "${OPENBABEL_SOURCE_DIR}/include/inchi")
add_executable(obengine src/engine.cpp)
target_include_directories(obengine PRIVATE "${OPENBABEL_SOURCE_DIR}/include" "${CMAKE_CURRENT_BINARY_DIR}/openbabel/include")
target_compile_definitions(obengine PRIVATE HAVE_STATIC_INCHI HAVE_EIGEN3)
target_link_libraries(obengine PRIVATE openbabel inchi)
if(UNIX AND NOT APPLE)
  target_link_options(obengine PRIVATE -static)
endif()
if(WIN32)
  add_executable(OpenBabelCN WIN32 src/gui.cpp src/app.rc)
  target_compile_definitions(OpenBabelCN PRIVATE UNICODE _UNICODE WINVER=0x0601 _WIN32_WINNT=0x0601)
  target_link_libraries(OpenBabelCN PRIVATE comctl32 comdlg32 shell32 ole32 uuid gdi32 user32 shlwapi)
endif()
if(MINGW)
  target_link_options(obengine PRIVATE -static -static-libgcc -static-libstdc++ -s)
  target_link_options(OpenBabelCN PRIVATE -municode -static -static-libgcc -static-libstdc++ -s)
endif()
if(MSVC)
  target_compile_options(obengine PRIVATE /utf-8)
  target_compile_options(OpenBabelCN PRIVATE /utf-8)
endif()
